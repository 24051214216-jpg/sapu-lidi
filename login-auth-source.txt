class _LoginPageState extends State<LoginPage> {

  final _formKey = GlobalKey<FormState>();
  final _phoneController = TextEditingController();
  final _passwordController = TextEditingController();

  late final Dio _dio;
  late final AuthRemoteDataSource _authDataSource;

  bool _rememberMe = false;
  bool _obscurePassword = true;
  bool _isLoading = false;

  @override
  void initState() {
    super.initState();

  @override
  void dispose() {
    _phoneController.dispose();

  void _showMessage(String message) {
    if (!mounted) return;

    ScaffoldMessenger.of(context)
      ..hideCurrentSnackBar()
      ..showSnackBar(
        SnackBar(
          content: Text(message),
          behavior: SnackBarBehavior.floating,
        ),
      );
  }

  String _normalizePhone(String value) {
    var phone = value.replaceAll(RegExp(r'\D'), '');

    if (phone.startsWith('0')) {
      phone = phone.substring(1);
    } else if (phone.startsWith('62')) {
      phone = phone.substring(2);
    }

    return '+62$phone';
  }

  Future<void> _handleLogin() async {
    FocusScope.of(context).unfocus();

    if (!_formKey.currentState!.validate()) return;

    setState(() => _isLoading = true);

    try {
      final phone = _normalizePhone(
        _phoneController.text,
      );

      final result = await _authDataSource.login(
        noHp: phone,
        password: _passwordController.text,
      );

      if (!mounted) return;

      final verified = await Navigator.push<bool>(
        context,
        MaterialPageRoute(
          builder: (_) => OtpVerificationPage(
            challengeId: result['challengeId'] as String,
            phoneNumber: result['phoneMasked'] as String,
            expiresIn: int.tryParse(
                  result['expiresIn'].toString(),
                ) ??
                300,
          ),
        ),
      );

      if (!mounted) return;

      if (verified == true) {
        _showMessage(
          'Verifikasi berhasil. Sesi autentikasi telah disimpan.',
        );
      }
    } on DioException catch (error) {
      final responseData = error.response?.data;

      final message =
          responseData is Map ? responseData['message']?.toString() : null;

      _showMessage(
        message ?? 'Tidak dapat terhubung ke server.',
      );
    } catch (error) {
      _showMessage(
        error.toString().replaceFirst('Exception: ', ''),
      );
    } finally {
      if (mounted) {
        setState(() => _isLoading = false);
      }
    }
  }

  void _openForgotPassword() {
    Navigator.push(
      context,
      MaterialPageRoute(
        builder: (_) => const ForgotPasswordPage(),
      ),
    );
  }

  void _openRegister() {

  }

  String? _validatePhone(String? value) {
    final phone = value?.replaceAll(RegExp(r'\D'), '') ?? '';

    if (phone.isEmpty) {
      return 'Nomor HP wajib diisi';
    }

    if (phone.length < 9 || phone.length > 13) {
      return 'Masukkan nomor HP yang valid';
    }

    return null;
  }

  String? _validatePassword(String? value) {
    if (value == null || value.isEmpty) {
      return 'Password wajib diisi';
    }

    if (value.length < 8) {
      return 'Password minimal 8 karakter';
    }

    return null;
  }

class RegisterPage extends StatefulWidget {
  const RegisterPage({super.key});

  @override
  State<RegisterPage> createState() => _RegisterPageState();
}

class _RegisterPageState extends State<RegisterPage> {
  static const Color _green = Color(0xFF006130);
  static const Color _greenLight = Color(0xFFEAF7ED);
  static const Color _text = Color(0xFF131B2E);
  static const Color _muted = Color(0xFF667085);
  static const Color _outline = Color(0xFFBECABD);

  final _formKey = GlobalKey<FormState>();

  final _nameController = TextEditingController();
  final _phoneController = TextEditingController();
  final _houseNumberController = TextEditingController();
  final _passwordController = TextEditingController();
  final _confirmPasswordController = TextEditingController();

  late final Dio _dio;
  late final AuthRemoteDataSource _authRemoteDataSource;
  final SecureStorage _secureStorage = SecureStorage();

  String? _selectedBlock;

  bool _obscurePassword = true;
  bool _obscureConfirmPassword = true;
  bool _agreementAccepted = false;
  bool _isLoading = false;

  static const Map<String, String> _blocks = {
    'A': 'Gang 1',
    'B': 'Gang 2-3',
    'C': 'Taman',
    'D': 'Gerbang Timur',
  };

  @override
  void initState() {
    super.initState();
    _dio = DioClient.create();
    _authRemoteDataSource = AuthRemoteDataSource(_dio);
    _passwordController.addListener(_refreshPasswordStrength);
  }

  @override
  void dispose() {
    _passwordController.removeListener(_refreshPasswordStrength);
    _nameController.dispose();
    _phoneController.dispose();
    _houseNumberController.dispose();
    _passwordController.dispose();
    _confirmPasswordController.dispose();
    super.dispose();
  }

  void _refreshPasswordStrength() {
    if (!mounted) return;

    setState(() {});

    if (_confirmPasswordController.text.isNotEmpty) {
      _formKey.currentState?.validate();
    }
  }

  int get _passwordStrength {
    final password = _passwordController.text;

    if (password.isEmpty) return 0;

    var score = 0;

    if (password.length >= 8) score++;
    if (RegExp(r'[A-Z]').hasMatch(password)) score++;
    if (RegExp(r'[a-z]').hasMatch(password)) score++;
    if (RegExp(r'[0-9]').hasMatch(password)) score++;
    if (RegExp(r'[^A-Za-z0-9]').hasMatch(password)) score++;

    if (score <= 2) return 1;
    if (score <= 3) return 2;

    return 3;
  }

  String get _passwordStrengthLabel {
    switch (_passwordStrength) {
      case 0:
        return 'Belum diisi';
      case 1:
        return 'Lemah';
      case 2:
        return 'Sedang';
      default:
        return 'Kuat';
    }
  }

  Color get _passwordStrengthColor {
    switch (_passwordStrength) {
      case 0:
        return const Color(0xFFD9DED9);
      case 1:
        return const Color(0xFFE05252);
      case 2:
        return const Color(0xFFEF9B27);
      default:
        return _green;
    }
  }

  String? _validateName(String? value) {
    final name = value?.trim() ?? '';

    if (name.isEmpty) {
      return 'Nama lengkap wajib diisi';
    }

    if (name.length < 3) {
      return 'Nama minimal 3 karakter';
    }

    if (RegExp(r'[0-9]').hasMatch(name)) {
      return 'Nama tidak boleh mengandung angka';
    }

    return null;
  }

  String? _validatePhone(String? value) {
    final phone = (value ?? '').replaceAll(RegExp(r'\D'), '');

    if (phone.isEmpty) {
      return 'Nomor WhatsApp wajib diisi';
    }

    if (phone.startsWith('0')) {
      return 'Hilangkan angka 0 setelah kode +62';
    }

    if (phone.length < 9 || phone.length > 13) {
      return 'Nomor HP harus terdiri dari 9-13 digit';
    }

    return null;
  }

  String? _validateBlock(String? value) {
    if (value == null || value.isEmpty) {
      return 'Silakan pilih blok rumah';
    }

    return null;
  }

  String? _validateHouseNumber(String? value) {
    final houseNumber = value?.trim() ?? '';

    if (houseNumber.isEmpty) {
      return 'Nomor rumah wajib diisi';
    }

    if (houseNumber.length > 15) {
      return 'Nomor rumah maksimal 15 karakter';
    }

    return null;
  }

  String? _validatePassword(String? value) {
    if (value == null || value.isEmpty) {
      return 'Password wajib diisi';
    }

    if (value.length < 8) {
      return 'Password minimal 8 karakter';
    }

    return null;
  }

  String? _validateConfirmPassword(String? value) {
    if (value == null || value.isEmpty) {
      return 'Ulangi password Anda';
    }

    if (value != _passwordController.text) {
      return 'Password tidak sama';
    }

    return null;
  }

  String _normalizePhoneNumber(String value) {
    var digits = value.replaceAll(RegExp(r'\D'), '');

    if (digits.startsWith('0')) {
      digits = digits.substring(1);
    }

    if (digits.startsWith('62')) {
      digits = digits.substring(2);
    }

    return '+62$digits';
  }

  void _showMessage(String message) {
    if (!mounted) return;

    ScaffoldMessenger.of(context)
      ..hideCurrentSnackBar()
      ..showSnackBar(
        SnackBar(
          content: Text(message),
          behavior: SnackBarBehavior.floating,
        ),
      );
  }

  Future<void> _handleRegister() async {
    FocusScope.of(context).unfocus();

    final isValid = _formKey.currentState?.validate() ?? false;

    if (!isValid) return;

    if (!_agreementAccepted) {
      _showMessage('Anda harus menyetujui syarat dan ketentuan.');
      return;
    }

    setState(() => _isLoading = true);

    try {
      final session = await _authRemoteDataSource.register(
        nama: _nameController.text.trim(),
        noHp: _normalizePhoneNumber(_phoneController.text),
        blokRumah: _selectedBlock!,
        noRumah: _houseNumberController.text.trim(),
        password: _passwordController.text,
      );

      final role = session.user['role']?.toString().toUpperCase();
      final status = session.user['status']?.toString().toUpperCase();

      if (role != 'WARGA' || status != 'AKTIF') {
        throw Exception(
          'Pendaftaran tersimpan, tetapi server tidak mengembalikan '
          'status akun AKTIF. Silakan hubungi administrator.',
        );
      }

      if (session.accessToken.isEmpty) {
        throw Exception(
          'Pendaftaran berhasil, tetapi access token tidak diterima.',
        );
      }

      await _secureStorage.saveAccessToken(session.accessToken);

import 'package:dio/dio.dart';

class OtpVerificationPage extends StatefulWidget {
  const OtpVerificationPage({
    super.key,
    required this.challengeId,
    required this.phoneNumber,
    this.expiresIn = 300,
  });

  final String challengeId;
  final String phoneNumber;
  final int expiresIn;

  @override
  State<OtpVerificationPage> createState() => _OtpVerificationPageState();
}

class _OtpVerificationPageState extends State<OtpVerificationPage> {
  static const Color _green = Color(0xFF006130);

  final _otpController = TextEditingController();

  late final Dio _dio;
  late final AuthRemoteDataSource _authDataSource;

  Timer? _timer;

  int _remainingSeconds = 300;
  bool _isVerifying = false;
  bool _isResending = false;

  @override
  void initState() {
    super.initState();

    _dio = DioClient.create();
    _authDataSource = AuthRemoteDataSource(_dio);

    _remainingSeconds = widget.expiresIn > 0 ? widget.expiresIn : 300;

    _startTimer();
  }

  @override
  void dispose() {
    _timer?.cancel();
    _otpController.dispose();
    super.dispose();
  }

  void _startTimer() {
    _timer?.cancel();

    _timer = Timer.periodic(
      const Duration(seconds: 1),
      (timer) {
        if (!mounted) {
          timer.cancel();
          return;
        }

        if (_remainingSeconds <= 1) {
          timer.cancel();

          setState(() {
            _remainingSeconds = 0;
          });
        } else {
          setState(() {
            _remainingSeconds--;
          });
        }
      },
    );
  }

  void _showMessage(String message) {
    if (!mounted) return;

    ScaffoldMessenger.of(context)
      ..hideCurrentSnackBar()
      ..showSnackBar(
        SnackBar(
          content: Text(message),
          behavior: SnackBarBehavior.floating,
        ),
      );
  }

  Future<void> _verifyOtp() async {
    FocusScope.of(context).unfocus();

    final otp = _otpController.text.trim();

    if (!RegExp(r'^\d{6}$').hasMatch(otp)) {
      _showMessage('Masukkan kode OTP 6 digit.');
      return;
    }

    setState(() => _isVerifying = true);

    try {
      final session = await _authDataSource.verifyOtp(
        challengeId: widget.challengeId,
        otp: otp,
      );

      await SecureStorage().saveAccessToken(
        session.accessToken,
      );

      if (!mounted) return;

      Navigator.pop(context, true);
    } on DioException catch (error) {
      final message = error.response?.data is Map
          ? error.response?.data['message']?.toString()
          : null;

      _showMessage(
        message ?? 'Tidak dapat menghubungi server.',
      );
    } catch (error) {
      _showMessage(
        error.toString().replaceFirst('Exception: ', ''),
      );
    } finally {
      if (mounted) {
        setState(() => _isVerifying = false);
      }
    }
  }

  Future<void> _resendOtp() async {
    setState(() => _isResending = true);

    try {
      await _authDataSource.requestOtp(
        challengeId: widget.challengeId,
      );

      if (!mounted) return;

      setState(() {
        _remainingSeconds = 300;
      });

      _startTimer();

      _showMessage('Kode OTP berhasil dikirim ulang.');
    } on DioException catch (error) {
      final message = error.response?.data is Map
          ? error.response?.data['message']?.toString()
          : null;

      _showMessage(
        message ?? 'Gagal mengirim ulang kode OTP.',

import 'package:dio/dio.dart';

import '../../../../core/constants/api_endpoints.dart';
import '../models/auth_session_model.dart';

class AuthRemoteDataSource {
  AuthRemoteDataSource(this._dio);

  final Dio _dio;

  Future<AuthSessionModel> register({
    required String nama,
    required String noHp,
    required String blokRumah,
    required String noRumah,
    required String password,
  }) async {
    final response = await _dio.post(
      ApiEndpoints.register,
      data: {
        'nama': nama,
        'noHp': noHp,
        'blokRumah': blokRumah,
        'noRumah': noRumah,
        'password': password,
      },
    );

    final body = _readBody(response.data);
    final data = _readData(body);

    if (body['success'] != true) {
      throw Exception(
        body['message']?.toString() ?? 'Pendaftaran warga gagal.',
      );
    }

    final session = AuthSessionModel.fromJson(data);

    if (session.accessToken.isEmpty) {
      throw Exception(
        'Server tidak mengembalikan access token setelah registrasi.',
      );
    }

    return session;
  }

  Future<Map<String, dynamic>> login({
    required String noHp,
    required String password,
  }) async {
    final response = await _dio.post(
      ApiEndpoints.login,
      data: {
        'noHp': noHp,
        'password': password,
      },
    );

    final body = _readBody(response.data);
    final data = _readData(body);

    final challengeId = data['challengeId']?.toString();

    if (body['success'] != true || challengeId == null || challengeId.isEmpty) {
      throw Exception(
        body['message']?.toString() ??
            'Respons login tidak memiliki challenge OTP.',
      );
    }

    return {
      'challengeId': challengeId,
      'phoneMasked': data['phoneMasked']?.toString() ?? noHp,
      'expiresIn': data['expiresIn'] ?? 300,
    };
  }

  Future<void> requestOtp({
    required String challengeId,
  }) async {
    final response = await _dio.post(
      ApiEndpoints.requestOtp,
      data: {
        'challengeId': challengeId,
      },
    );

    final body = _readBody(response.data);

    if (body['success'] != true) {
      throw Exception(
        body['message']?.toString() ?? 'Gagal mengirim ulang kode OTP.',
      );
    }
  }

  Future<AuthSessionModel> verifyOtp({
    required String challengeId,
    required String otp,
  }) async {
    final response = await _dio.post(
      ApiEndpoints.verifyOtp,
      data: {
        'challengeId': challengeId,
        'otp': otp,
      },
    );

    final body = _readBody(response.data);
    final data = _readData(body);

    if (body['success'] != true) {
      throw Exception(
        body['message']?.toString() ?? 'Verifikasi OTP gagal.',
      );
    }

    final session = AuthSessionModel.fromJson(data);

    if (session.accessToken.isEmpty) {
      throw Exception(
        'Access token tidak ditemukan setelah verifikasi OTP.',
      );
    }

    return session;
  }

  Map<String, dynamic> _readBody(dynamic responseData) {
    if (responseData is Map<String, dynamic>) {
      return responseData;
    }

    throw Exception('Format respons server tidak valid.');
  }

  Map<String, dynamic> _readData(
    Map<String, dynamic> body,
  ) {
    final data = body['data'];

    if (data is Map<String, dynamic>) {
      return data;
    }

    throw Exception('Data respons server tidak ditemukan.');
  }
}

export async function registerUser(data) {
  const existingUser = await findUserByPhone(data.noHp);

  if (existingUser && !existingUser.deletedAt) {
    throw new AppError('Nomor WhatsApp sudah terdaftar.', 409);
  }

  if (existingUser?.deletedAt) {
    throw new AppError(
      'Nomor WhatsApp pernah digunakan pada akun yang dihapus. Hubungi administrator.',
      409,
    );
  }

  const password = await hashPassword(data.password);

  try {
    const user = await createUser({
      ...data,
      password,
    });

    return createAuthResult(user);
  } catch (error) {
    if (error.code === '23505') {
      throw new AppError('Nomor WhatsApp sudah terdaftar.', 409);
    }

    throw error;
  }
}

export async function loginUser(data) {
  const user = await findUserByPhone(data.noHp);

  if (!user || user.deletedAt) {
    throw new AppError('Nomor WhatsApp atau password salah.', 401);
  }

  const passwordMatches = await comparePassword(
    data.password,
    user.password,
  );

  if (!passwordMatches) {
    throw new AppError('Nomor WhatsApp atau password salah.', 401);
  }

  if (!isActiveUser(user)) {
    throw new AppError('Akun belum aktif atau tidak dapat digunakan.', 403);
  }

  return createAuthResult(user);
}

export async function requestOtp(data) {
  const user = await findUserByPhone(data.noHp);

  if (!isActiveUser(user)) {
    throw new AppError(
      'Akun tidak ditemukan atau belum aktif.',
      404,
    );
  }

  if (!Object.values(OTP_PURPOSE).includes(data.purpose)) {
    throw new AppError('Tujuan OTP tidak valid.', 400);
  }

  const client = await pool.connect();
  let otpData;

  try {
    await client.query('BEGIN');

    const latestOtp = await findLatestActiveOtp(
      data.noHp,
      data.purpose,
      client,
    );

    if (latestOtp) {
      const elapsedSeconds =
        (Date.now() - new Date(latestOtp.createdAt).getTime()) / 1000;

      if (elapsedSeconds < 60) {
        throw new AppError(
          'Mohon tunggu 60 detik sebelum meminta OTP kembali.',
          429,
        );
      }
    }

    await consumePreviousOtps(
      data.noHp,
      data.purpose,
      client,
    );

    otpData = createOtpForPhone(data.noHp, data.purpose);

    await createOtp({
      phone: data.noHp,
      purpose: data.purpose,
      otpHash: otpData.otpHash,
      expiresAt: otpData.expiresAt,
    }, client);

    await client.query('COMMIT');
  } catch (error) {
    await client.query('ROLLBACK').catch(() => {});
    throw error;
  } finally {
    client.release();
  }

  try {
    await sendWhatsAppOtp(data.noHp, otpData.otp);
  } catch (error) {
    await pool.query(
      `UPDATE public."AuthOtp"
       SET "consumedAt" = CURRENT_TIMESTAMP
       WHERE "noHp" = $1
         AND purpose = $2
         AND "consumedAt" IS NULL`,
      [data.noHp, data.purpose],
    );

    throw new AppError(
      'OTP gagal dikirim. Silakan coba kembali beberapa saat lagi.',
      502,
    );
  }

  return {
    message: 'OTP berhasil dikirim.',
    expiresIn: env.otp.expiresMinutes * 60,
  };
}

export async function verifyOtp(data) {
  const client = await pool.connect();
  let outcome;

  try {
    await client.query('BEGIN');

    const otpResult = await client.query(
      `SELECT *
       FROM public."AuthOtp"
       WHERE "noHp" = $1
         AND purpose = $2
         AND "consumedAt" IS NULL
       ORDER BY "createdAt" DESC
       LIMIT 1
       FOR UPDATE`,
      [data.noHp, data.purpose],
    );

    const otpRecord = otpResult.rows[0];

    if (!otpRecord) {
      throw new AppError(
        'OTP tidak ditemukan atau sudah digunakan.',
        400,
      );
    }

    if (isOtpExpired(otpRecord)) {
      await consumeOtp(otpRecord.id, client);
      outcome = {
        error: new AppError('OTP sudah kedaluwarsa.', 400),
      };
    } else if (hasOtpExceededAttempts(otpRecord)) {
      await consumeOtp(otpRecord.id, client);
      outcome = {
        error: new AppError(
          'Batas percobaan OTP telah tercapai. Minta OTP baru.',
          429,
        ),
      };
    } else {
      const submittedHash = hashOtp(
        data.noHp,
        data.purpose,
        data.otp,
      );

      if (!safeCompareHash(otpRecord.otpHash, submittedHash)) {
        await incrementOtpAttempts(otpRecord.id, client);

        outcome = {
          error: new AppError('Kode OTP salah.', 400),
        };
      } else {
        await consumeOtp(otpRecord.id, client);

        const user = await findUserByPhone(data.noHp, client);

        if (!isActiveUser(user)) {
          outcome = {
            error: new AppError(
              'Akun tidak ditemukan atau belum aktif.',
              403,
            ),
          };
        } else {
          outcome = { user };
        }
      }
    }

    await client.query('COMMIT');
  } catch (error) {
    await client.query('ROLLBACK').catch(() => {});
    throw error;
  } finally {
    client.release();
  }

  if (outcome.error) {
    throw outcome.error;
  }

  if (data.purpose === OTP_PURPOSE.LOGIN) {
    return createAuthResult(outcome.user);
  }

  return {
    resetToken: createResetToken(outcome.user),
    tokenType: 'Bearer',
    expiresIn: '10m',
  };
}

export async function resetPassword(data) {
  let payload;

  try {
    payload = verifyResetToken(data.resetToken);
  } catch {
    throw new AppError(
      'Reset token tidak valid atau sudah kedaluwarsa.',
      401,
    );
  }

  if (
    payload.type !== 'password_reset'
    || !payload.sub
    || !payload.passwordFingerprint
  ) {
    throw new AppError('Reset token tidak valid.', 401);
  }

  const user = await findUserById(payload.sub);

  if (!isActiveUser(user)) {
    throw new AppError('Akun tidak ditemukan atau tidak aktif.', 404);
  }

  const currentFingerprint = createPasswordFingerprint(user.password);

  if (currentFingerprint !== payload.passwordFingerprint) {
    throw new AppError(
      'Reset token sudah digunakan atau tidak berlaku.',
      401,
    );
  }

  const newHashedPassword = await hashPassword(data.newPassword);

  const updated = await updateUserPassword(
    user.id,
    newHashedPassword,
  );

  if (!updated) {
    throw new AppError('Gagal memperbarui password.', 500);
  }

  return {
    message: 'Password berhasil diperbarui.',
  };
}

    throw new AppError(
      'Pengguna tidak ditemukan atau akun tidak aktif.',
      404,
    );
  }

  return publicUser(user);
}

function validate(schema, data) {
  const result = schema.safeParse(data);

  if (!result.success) {
    const error = new Error(
      result.error.issues.map((item) => item.message).join(', '),
    );

    error.statusCode = 400;
    throw error;
  }

  return result.data;
}

function sendError(res, error) {
  const statusCode = Number.isInteger(error.statusCode)
    ? error.statusCode
    : 500;

  if (statusCode >= 500) {
    console.error(error);
  }

  return res.status(statusCode).json({
    success: false,
    message: statusCode >= 500
      ? 'Terjadi kesalahan pada server.'
      : error.message,
  });
}

  rt,

`;

export async function findUserByPhone(phone, client = pool) {
  const result = await client.query(
    `SELECT ${USER_COLUMNS}, password, "deletedAt"
     FROM public."User"
     WHERE "noHp" = $1
     LIMIT 1`,
    [phone],
  );

  return result.rows[0] || null;
}

export async function findUserById(id, client = pool) {
  const result = await client.query(
    `SELECT ${USER_COLUMNS}, password, "deletedAt"
     FROM public."User"
     WHERE id = $1
     LIMIT 1`,
    [id],
  );

  return result.rows[0] || null;
}

export async function createUser(data, client = pool) {
  const result = await client.query(
    `INSERT INTO public."User" (
      id,
      nama,
      "noHp",
      password,
      role,
      status,
      "blokRumah",
      "noRumah",
      "createdAt",
      "updatedAt"
    )
    VALUES (
      gen_random_uuid(),
      $1,
      $2,
      $3,
      'WARGA',
      'AKTIF',
      $4,
      $5,
      CURRENT_TIMESTAMP,
      CURRENT_TIMESTAMP
    )
    RETURNING ${USER_COLUMNS}`,
    [
      data.nama,
      data.noHp,
      data.password,
      data.blokRumah,
      data.noRumah,
    ],
  );

  return result.rows[0];
}

export async function updateUserPassword(id, password, client = pool) {
  const result = await client.query(
    `UPDATE public."User"
     SET password = $1,
         "updatedAt" = CURRENT_TIMESTAMP
     WHERE id = $2
       AND "deletedAt" IS NULL
     RETURNING id`,
    [password, id],
  );

  return result.rowCount > 0;
}

import { z } from 'zod';
import {
  isValidPhoneNumber,
  normalizePhoneNumber,
} from '../utils/phone.js';

const phoneSchema = z.string()
  .trim()
  .min(1, 'Nomor HP wajib diisi.')
  .max(30, 'Nomor HP terlalu panjang.')
  .transform(normalizePhoneNumber)
  .refine(
    isValidPhoneNumber,
    'Nomor WhatsApp Indonesia tidak valid.',
  );

export const registerSchema = z.object({
  nama: z.string()
    .trim()
    .min(3, 'Nama minimal 3 karakter.')
    .max(100, 'Nama maksimal 100 karakter.'),

  noHp: phoneSchema,

  blokRumah: z.string()
    .trim()
    .min(1, 'Blok rumah wajib diisi.')
    .max(30),

  noRumah: z.string()
    .trim()
    .min(1, 'Nomor rumah wajib diisi.')
    .max(30),

  password: z.string()
    .min(8, 'Password minimal 8 karakter.')
    .max(72, 'Password maksimal 72 karakter.'),
});

export const loginSchema = z.object({
  noHp: phoneSchema,
  password: z.string().min(1, 'Password wajib diisi.'),
});

export const otpRequestSchema = z.object({
  noHp: phoneSchema,
  purpose: z.enum(['LOGIN', 'RESET_PASSWORD']),
});

export const otpVerifySchema = z.object({
  noHp: phoneSchema,
  purpose: z.enum(['LOGIN', 'RESET_PASSWORD']),
  otp: z.string()
    .regex(/^[0-9]{6}$/, 'OTP harus terdiri dari 6 digit.'),
});

export const resetPasswordSchema = z.object({
  resetToken: z.string().min(1, 'Reset token wajib diisi.'),
  newPassword: z.string()
    .min(8, 'Password minimal 8 karakter.')
    .max(72, 'Password maksimal 72 karakter.'),
});
