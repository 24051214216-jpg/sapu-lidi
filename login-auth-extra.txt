import { env } from '../config/env.config.js';
import {
  generateOtp,
  hashOtp,
} from '../utils/hash.js';

export function createOtpForPhone(phone, purpose) {
  const otp = generateOtp();

  const expiresAt = new Date(
    Date.now() + env.otp.expiresMinutes * 60 * 1000,
  );

  return {
    otp,
    otpHash: hashOtp(phone, purpose, otp),
    expiresAt,
  };
}

export function isOtpExpired(otpRecord) {
  return new Date(otpRecord.expiresAt).getTime() <= Date.now();
}

export function hasOtpExceededAttempts(otpRecord) {
  return otpRecord.attempts >= env.otp.maxAttempts;
}

import { pool } from '../config/database.js';

export async function findLatestActiveOtp(phone, purpose, client = pool) {
  const result = await client.query(
    `SELECT *
     FROM public."AuthOtp"
     WHERE "noHp" = $1
       AND purpose = $2
       AND "consumedAt" IS NULL
     ORDER BY "createdAt" DESC
     LIMIT 1`,
    [phone, purpose],
  );

  return result.rows[0] || null;
}

export async function createOtp({
  phone,
  purpose,
  otpHash,
  expiresAt,
}, client = pool) {
  const result = await client.query(
    `INSERT INTO public."AuthOtp" (
      id,
      "noHp",
      purpose,
      "otpHash",
      "expiresAt",
      attempts,
      "createdAt"
    )
    VALUES (
      gen_random_uuid(),
      $1,
      $2,
      $3,
      $4,
      0,
      CURRENT_TIMESTAMP
    )
    RETURNING id, "noHp", purpose, "expiresAt", "createdAt"`,
    [phone, purpose, otpHash, expiresAt],
  );

  return result.rows[0];
}

export async function consumeOtp(id, client = pool) {
  await client.query(
    `UPDATE public."AuthOtp"
     SET "consumedAt" = CURRENT_TIMESTAMP
     WHERE id = $1
       AND "consumedAt" IS NULL`,
    [id],
  );
}

export async function incrementOtpAttempts(id, client = pool) {
  await client.query(
    `UPDATE public."AuthOtp"
     SET attempts = attempts + 1
     WHERE id = $1`,
    [id],
  );
}

export async function consumePreviousOtps(phone, purpose, client = pool) {
  await client.query(
    `UPDATE public."AuthOtp"
     SET "consumedAt" = CURRENT_TIMESTAMP
     WHERE "noHp" = $1
       AND purpose = $2
       AND "consumedAt" IS NULL`,
    [phone, purpose],
  );
}

import jwt from 'jsonwebtoken';

import { jwtConfig } from '../config/jwt.config.js';
import { createPasswordFingerprint } from './hash.js';

export function createAccessToken(user) {
  return jwt.sign(
    {
      role: user.role,
      type: 'access',
    },
    jwtConfig.accessSecret,
    {
      subject: user.id,
      expiresIn: jwtConfig.accessExpiresIn,
    },
  );
}

export function verifyAccessToken(token) {
  return jwt.verify(token, jwtConfig.accessSecret);
}

export function createResetToken(user) {
  return jwt.sign(
    {
      type: 'password_reset',
      passwordFingerprint: createPasswordFingerprint(user.password),
    },
    jwtConfig.resetSecret,
    {
      subject: user.id,
      expiresIn: jwtConfig.resetExpiresIn,
    },
  );
}

export function verifyResetToken(token) {
  return jwt.verify(token, jwtConfig.resetSecret);
}


========== backend/src/utils/hash.js ==========

import bcrypt from 'bcryptjs';
import crypto from 'node:crypto';

import { env } from '../config/env.config.js';

const BCRYPT_ROUNDS = 12;

export async function hashPassword(password) {
  return bcrypt.hash(password, BCRYPT_ROUNDS);
}

export async function comparePassword(password, hashedPassword) {
  return bcrypt.compare(password, hashedPassword);
}

export function hashOtp(phone, purpose, otp) {
  return crypto
    .createHmac('sha256', env.otp.hashSecret)
    .update(`${phone}:${purpose}:${otp}`)
    .digest('hex');
}

export function generateOtp() {
  return crypto.randomInt(0, 1000000).toString().padStart(6, '0');
}

export function safeCompareHash(firstHash, secondHash) {
  if (
    typeof firstHash !== 'string'
    || typeof secondHash !== 'string'
    || !/^[a-f0-9]{64}$/i.test(firstHash)
    || !/^[a-f0-9]{64}$/i.test(secondHash)
  ) {
    return false;
  }

  return crypto.timingSafeEqual(
    Buffer.from(firstHash, 'hex'),
    Buffer.from(secondHash, 'hex'),
  );
}

export function createPasswordFingerprint(passwordHash) {
  return crypto
    .createHash('sha256')
    .update(passwordHash)
    .digest('hex');
}

========== frontend/lib/features/auth/data/models/auth_session_model.dart ==========

class AuthSessionModel {
  const AuthSessionModel({
    required this.accessToken,
    required this.tokenType,
    required this.user,
  });

  final String accessToken;
  final String tokenType;
  final Map<String, dynamic> user;

  factory AuthSessionModel.fromJson(
    Map<String, dynamic> json,
  ) {
    final dynamic userData = json['user'];

    return AuthSessionModel(
      accessToken: json['accessToken']?.toString() ?? '',
      tokenType: json['tokenType']?.toString() ?? 'Bearer',
      user: userData is Map
          ? Map<String, dynamic>.from(userData)
          : <String, dynamic>{},
    );
  }

  Map<String, dynamic> toJson() {
    return {
      'accessToken': accessToken,
      'tokenType': tokenType,
      'user': user,
    };
  }

  bool get hasValidToken => accessToken.isNotEmpty;

  AuthSessionModel copyWith({
    String? accessToken,
    String? tokenType,
    Map<String, dynamic>? user,
  }) {
    return AuthSessionModel(
      accessToken: accessToken ?? this.accessToken,
      tokenType: tokenType ?? this.tokenType,
      user: user ?? this.user,
    );
  }
}
