function isActiveUser(user) {
  return Boolean(
    user
    && !user.deletedAt
    && user.status === USER_STATUS.AKTIF,
  );
}

function publicUser(user) {
  return {
    id: user.id,
    nama: user.nama,
    noHp: user.noHp,
    role: user.role,
    status: user.status,
    blokRumah: user.blokRumah,
    noRumah: user.noRumah,
    fotoProfil: user.fotoProfil,
    rt: user.rt,
    rw: user.rw,
  };
}

function createAuthResult(user) {
  return {
    user: publicUser(user),
    accessToken: createAccessToken(user),
    tokenType: 'Bearer',
    expiresIn: env.jwt.expiresIn,
  };
}

export async function registerUser(data) {
  const existingUser = await findUserByPhone(data.noHp);

  if (existingUser && !existingUser.deletedAt) {
    throw new AppError('Nomor WhatsApp sudah terdaftar.', 409);
  }

  if (existingUser?.deletedAt) {
    throw new AppError(
      'Nomor WhatsApp pernah digunakan pada akun yang dihapus. Hubungi administrator.',
      409,
    );
  }

  const password = await hashPassword(data.password);

  try {
    const user = await createUser({
      ...data,
      password,
    });

    return createAuthResult(user);
  } catch (error) {
    if (error.code === '23505') {
      throw new AppError('Nomor WhatsApp sudah terdaftar.', 409);
    }

    throw error;
  }
}

export async function loginUser(data) {
  const user = await findUserByPhone(data.noHp);

  if (!user || user.deletedAt) {
    throw new AppError('Nomor WhatsApp atau password salah.', 401);
  }

  const passwordMatches = await comparePassword(
    data.password,
    user.password,
  );

  if (!passwordMatches) {
    throw new AppError('Nomor WhatsApp atau password salah.', 401);
  }

  if (!isActiveUser(user)) {
    throw new AppError('Akun belum aktif atau tidak dapat digunakan.', 403);
  }

  return createAuthResult(user);
}

export async function requestOtp(data) {
  const user = await findUserByPhone(data.noHp);

  if (!isActiveUser(user)) {
    throw new AppError(
      'Akun tidak ditemukan atau belum aktif.',
      404,
    );
  }

  if (!Object.values(OTP_PURPOSE).includes(data.purpose)) {
    throw new AppError('Tujuan OTP tidak valid.', 400);
  }

  const client = await pool.connect();
  let otpData;

  try {
    await client.query('BEGIN');

    const latestOtp = await findLatestActiveOtp(
      data.noHp,
      data.purpose,
      client,
    );

    if (latestOtp) {
      const elapsedSeconds =
        (Date.now() - new Date(latestOtp.createdAt).getTime()) / 1000;

      if (elapsedSeconds < 60) {
        throw new AppError(
          'Mohon tunggu 60 detik sebelum meminta OTP kembali.',
          429,
        );
      }
    }

    await consumePreviousOtps(
      data.noHp,
      data.purpose,
      client,
    );

    otpData = createOtpForPhone(data.noHp, data.purpose);

    await createOtp({
      phone: data.noHp,
      purpose: data.purpose,
      otpHash: otpData.otpHash,
      expiresAt: otpData.expiresAt,
    }, client);

    await client.query('COMMIT');
  } catch (error) {
    await client.query('ROLLBACK').catch(() => {});
    throw error;
  } finally {
    client.release();
  }

  try {
    await sendWhatsAppOtp(data.noHp, otpData.otp);
  } catch (error) {
    await pool.query(
      `UPDATE public."AuthOtp"
       SET "consumedAt" = CURRENT_TIMESTAMP
       WHERE "noHp" = $1
         AND purpose = $2
         AND "consumedAt" IS NULL`,
      [data.noHp, data.purpose],
    );

    throw new AppError(
      'OTP gagal dikirim. Silakan coba kembali beberapa saat lagi.',
      502,
    );
  }

  return {
    message: 'OTP berhasil dikirim.',
    expiresIn: env.otp.expiresMinutes * 60,
  };
}

export async function verifyOtp(data) {
  const client = await pool.connect();
  let outcome;

  try {
    await client.query('BEGIN');

    const otpResult = await client.query(
      `SELECT *
       FROM public."AuthOtp"
       WHERE "noHp" = $1
         AND purpose = $2
         AND "consumedAt" IS NULL
       ORDER BY "createdAt" DESC
       LIMIT 1
       FOR UPDATE`,
      [data.noHp, data.purpose],
    );

    const otpRecord = otpResult.rows[0];

    if (!otpRecord) {
      throw new AppError(
        'OTP tidak ditemukan atau sudah digunakan.',
        400,
      );
    }

    if (isOtpExpired(otpRecord)) {
      await consumeOtp(otpRecord.id, client);
      outcome = {
        error: new AppError('OTP sudah kedaluwarsa.', 400),
      };
    } else if (hasOtpExceededAttempts(otpRecord)) {
      await consumeOtp(otpRecord.id, client);
      outcome = {
        error: new AppError(
          'Batas percobaan OTP telah tercapai. Minta OTP baru.',
          429,
        ),
      };
    } else {
      const submittedHash = hashOtp(
        data.noHp,
        data.purpose,
        data.otp,
      );

      if (!safeCompareHash(otpRecord.otpHash, submittedHash)) {
        await incrementOtpAttempts(otpRecord.id, client);

        outcome = {
          error: new AppError('Kode OTP salah.', 400),
        };
      } else {
        await consumeOtp(otpRecord.id, client);

        const user = await findUserByPhone(data.noHp, client);

        if (!isActiveUser(user)) {
          outcome = {
            error: new AppError(
              'Akun tidak ditemukan atau belum aktif.',
              403,
            ),
          };
        } else {
          outcome = { user };
        }
      }
    }

    await client.query('COMMIT');
  } catch (error) {
    await client.query('ROLLBACK').catch(() => {});
    throw error;
  } finally {
    client.release();
  }

  if (outcome.error) {
    throw outcome.error;
  }

  if (data.purpose === OTP_PURPOSE.LOGIN) {
    return createAuthResult(outcome.user);
  }

  return {
    resetToken: createResetToken(outcome.user),
    tokenType: 'Bearer',
    expiresIn: '10m',
  };
}

export async function resetPassword(data) {
  let payload;

  try {
    payload = verifyResetToken(data.resetToken);
  } catch {
    throw new AppError(
      'Reset token tidak valid atau sudah kedaluwarsa.',
      401,
    );
  }

  if (
    payload.type !== 'password_reset'
    || !payload.sub
    || !payload.passwordFingerprint
  ) {
    throw new AppError('Reset token tidak valid.', 401);
  }

  const user = await findUserById(payload.sub);

  if (!isActiveUser(user)) {
    throw new AppError('Akun tidak ditemukan atau tidak aktif.', 404);
  }

  const currentFingerprint = createPasswordFingerprint(user.password);

  if (currentFingerprint !== payload.passwordFingerprint) {
    throw new AppError(
      'Reset token sudah digunakan atau tidak berlaku.',
      401,
    );
  }

  const newHashedPassword = await hashPassword(data.newPassword);

  const updated = await updateUserPassword(
    user.id,
    newHashedPassword,
  );

  if (!updated) {
    throw new AppError('Gagal memperbarui password.', 500);
  }

  return {
    message: 'Password berhasil diperbarui.',
  };
}

export async function getCurrentUser(userId) {
  const user = await findUserById(userId);

  if (!isActiveUser(user)) {
    throw new AppError(
      'Pengguna tidak ditemukan atau akun tidak aktif.',
      404,
    );
  }

  return publicUser(user);
}

import { pool } from '../config/database.js';

export async function findLatestActiveOtp(phone, purpose, client = pool) {
  const result = await client.query(
    `SELECT *
     FROM public."AuthOtp"
     WHERE "noHp" = $1
       AND purpose = $2
       AND "consumedAt" IS NULL
     ORDER BY "createdAt" DESC
     LIMIT 1`,
    [phone, purpose],
  );

  return result.rows[0] || null;
}

export async function createOtp({
  phone,
  purpose,
  otpHash,
  expiresAt,
}, client = pool) {
  const result = await client.query(
    `INSERT INTO public."AuthOtp" (
      id,
      "noHp",
      purpose,
      "otpHash",
      "expiresAt",
      attempts,
      "createdAt"
    )
    VALUES (
      gen_random_uuid(),
      $1,
      $2,
      $3,
      $4,
      0,
      CURRENT_TIMESTAMP
    )
    RETURNING id, "noHp", purpose, "expiresAt", "createdAt"`,
    [phone, purpose, otpHash, expiresAt],
  );

  return result.rows[0];
}

export async function consumeOtp(id, client = pool) {
  await client.query(
    `UPDATE public."AuthOtp"
     SET "consumedAt" = CURRENT_TIMESTAMP
     WHERE id = $1
       AND "consumedAt" IS NULL`,
    [id],
  );
}

export async function incrementOtpAttempts(id, client = pool) {
  await client.query(
    `UPDATE public."AuthOtp"
     SET attempts = attempts + 1
     WHERE id = $1`,
    [id],
  );
}

export async function consumePreviousOtps(phone, purpose, client = pool) {
  await client.query(
    `UPDATE public."AuthOtp"
     SET "consumedAt" = CURRENT_TIMESTAMP
     WHERE "noHp" = $1
       AND purpose = $2
       AND "consumedAt" IS NULL`,
    [phone, purpose],
  );
}

function validate(schema, data) {
  const result = schema.safeParse(data);

  if (!result.success) {
    const error = new Error(
      result.error.issues.map((item) => item.message).join(', '),
    );

    error.statusCode = 400;
    throw error;
  }

  return result.data;
}

function sendError(res, error) {
  const statusCode = Number.isInteger(error.statusCode)
    ? error.statusCode
    : 500;

  if (statusCode >= 500) {
    console.error(error);
  }

  return res.status(statusCode).json({
    success: false,
    message: statusCode >= 500
      ? 'Terjadi kesalahan pada server.'
      : error.message,
  });
}

export async function register(req, res) {
  try {
    const data = validate(registerSchema, req.body);
    const result = await registerUser(data);

    return res.status(201).json({
      success: true,
      message: 'Registrasi berhasil.',
      data: result,
    });
  } catch (error) {
    return sendError(res, error);
  }
}

export function createOtpForPhone(phone, purpose) {
  const otp = generateOtp();

  const expiresAt = new Date(
    Date.now() + env.otp.expiresMinutes * 60 * 1000,
  );

  return {
    otp,
    otpHash: hashOtp(phone, purpose, otp),
    expiresAt,
  };

export function isOtpExpired(otpRecord) {
  return new Date(otpRecord.expiresAt).getTime() <= Date.now();
}

export function hasOtpExceededAttempts(otpRecord) {
  return otpRecord.attempts >= env.otp.maxAttempts;
}

import { env } from '../config/env.config.js';

export async function sendOtp(phone, otp) {
  const provider = process.env.OTP_PROVIDER || 'console';

  if (provider === 'console') {
    if (env.nodeEnv === 'production') {
      throw new Error(
        'Provider console tidak boleh digunakan di production.',
      );
    }

    console.log('================================');
    console.log('SAPULIDI OTP');
    console.log(`Nomor HP : ${phone}`);
    console.log(`Kode OTP : ${otp}`);
    console.log('================================');

    return { provider: 'console' };
  }

  if (provider !== 'meta') {
    throw new Error('OTP_PROVIDER tidak dikenal.');
  }

  const {
    WA_PHONE_NUMBER_ID,
    WA_ACCESS_TOKEN,
    WA_TEMPLATE_NAME,
    WA_TEMPLATE_LANGUAGE,
    WA_GRAPH_API_VERSION,
  } = process.env;

  if (
    !WA_PHONE_NUMBER_ID
    || !WA_ACCESS_TOKEN
    || !WA_TEMPLATE_NAME
    || !WA_GRAPH_API_VERSION
  ) {
    throw new Error('Konfigurasi WhatsApp Cloud API belum lengkap.');
  }

  const url =
    `https://graph.facebook.com/${WA_GRAPH_API_VERSION}`
    + `/${WA_PHONE_NUMBER_ID}/messages`;

  const response = await fetch(url, {
    method: 'POST',
    headers: {
      Authorization: `Bearer ${WA_ACCESS_TOKEN}`,
      'Content-Type': 'application/json',
    },
    body: JSON.stringify({
      messaging_product: 'whatsapp',
      to: phone.replace('+', ''),
      type: 'template',
      template: {
        name: WA_TEMPLATE_NAME,
        language: {
          code: WA_TEMPLATE_LANGUAGE || 'id',
        },
        components: [
          {
            type: 'body',
            parameters: [
              {
                type: 'text',
                text: otp,
              },
            ],
          },
        ],
      },
    }),
  });

  const result = await response.json().catch(() => ({}));

  if (!response.ok) {
    console.error('WhatsApp Cloud API gagal:', {
      status: response.status,
      error: result.error?.message || 'Unknown error',
    });

    throw new Error('Gagal mengirim OTP melalui WhatsApp.');
  }

  return result;
}

import 'package:dio/dio.dart';

import '../../../../core/constants/api_endpoints.dart';
import '../models/auth_session_model.dart';

class AuthRemoteDataSource {
  AuthRemoteDataSource(this._dio);

  final Dio _dio;

  Future<AuthSessionModel> register({
    required String nama,
    required String noHp,
    required String blokRumah,
    required String noRumah,
    required String password,
  }) async {
    final response = await _dio.post(
      ApiEndpoints.register,
      data: {
        'nama': nama,
        'noHp': noHp,
        'blokRumah': blokRumah,
        'noRumah': noRumah,
        'password': password,
      },
    );

    final body = _readBody(response.data);
    final data = _readData(body);

    if (body['success'] != true) {
      throw Exception(
        body['message']?.toString() ?? 'Pendaftaran warga gagal.',
      );
    }

    final session = AuthSessionModel.fromJson(data);

    if (session.accessToken.isEmpty) {
      throw Exception(
        'Server tidak mengembalikan access token setelah registrasi.',
      );
    }

    return session;
  }

  Future<Map<String, dynamic>> login({
    required String noHp,
    required String password,
  }) async {
    final response = await _dio.post(
      ApiEndpoints.login,
      data: {
        'noHp': noHp,
        'password': password,
      },
    );

    final body = _readBody(response.data);
    final data = _readData(body);

    final challengeId = data['challengeId']?.toString();

    if (body['success'] != true || challengeId == null || challengeId.isEmpty) {
      throw Exception(
        body['message']?.toString() ??
            'Respons login tidak memiliki challenge OTP.',
      );
    }

    return {
      'challengeId': challengeId,
      'phoneMasked': data['phoneMasked']?.toString() ?? noHp,
      'expiresIn': data['expiresIn'] ?? 300,
    };
  }

  Future<void> requestOtp({
    required String challengeId,
  }) async {
    final response = await _dio.post(
      ApiEndpoints.requestOtp,
      data: {
        'challengeId': challengeId,
      },
    );

    final body = _readBody(response.data);

    if (body['success'] != true) {
      throw Exception(
        body['message']?.toString() ?? 'Gagal mengirim ulang kode OTP.',
      );
    }
  }

  Future<AuthSessionModel> verifyOtp({
    required String challengeId,
    required String otp,
  }) async {
    final response = await _dio.post(
      ApiEndpoints.verifyOtp,
      data: {
        'challengeId': challengeId,
        'otp': otp,
      },
    );

    final body = _readBody(response.data);
    final data = _readData(body);

    if (body['success'] != true) {
      throw Exception(
        body['message']?.toString() ?? 'Verifikasi OTP gagal.',
      );
    }

    final session = AuthSessionModel.fromJson(data);

    if (session.accessToken.isEmpty) {
      throw Exception(
        'Access token tidak ditemukan setelah verifikasi OTP.',
      );
    }

    return session;
  }

  Map<String, dynamic> _readBody(dynamic responseData) {
    if (responseData is Map<String, dynamic>) {
      return responseData;
    }

    throw Exception('Format respons server tidak valid.');
  }

  Map<String, dynamic> _readData(
    Map<String, dynamic> body,
  ) {
    final data = body['data'];

    if (data is Map<String, dynamic>) {
      return data;
    }

    throw Exception('Data respons server tidak ditemukan.');
  }
}

class AuthSessionModel {
  const AuthSessionModel({
    required this.accessToken,
    required this.tokenType,
    required this.user,
  });

  final String accessToken;
  final String tokenType;
  final Map<String, dynamic> user;

  factory AuthSessionModel.fromJson(
    Map<String, dynamic> json,
  ) {
    final dynamic userData = json['user'];

    return AuthSessionModel(
      accessToken: json['accessToken']?.toString() ?? '',
      tokenType: json['tokenType']?.toString() ?? 'Bearer',
      user: userData is Map
          ? Map<String, dynamic>.from(userData)
          : <String, dynamic>{},
    );
  }

  void initState() {
    super.initState();

    _dio = DioClient.create();

  void _startTimer() {
    _timer?.cancel();

    _timer = Timer.periodic(
      const Duration(seconds: 1),
      (timer) {
        if (!mounted) {
          timer.cancel();
          return;
        }

        if (_remainingSeconds <= 1) {
          timer.cancel();

          setState(() {
            _remainingSeconds = 0;
          });
        } else {
          setState(() {
            _remainingSeconds--;
          });
        }
      },
    );
  }

  void _showMessage(String message) {
    if (!mounted) return;

    ScaffoldMessenger.of(context)
      ..hideCurrentSnackBar()

          behavior: SnackBarBehavior.floating,
        ),
      );
  }

  Future<void> _verifyOtp() async {
    FocusScope.of(context).unfocus();

    final otp = _otpController.text.trim();

    if (!RegExp(r'^\d{6}$').hasMatch(otp)) {
      _showMessage('Masukkan kode OTP 6 digit.');
      return;
    }

    setState(() => _isVerifying = true);

    try {
      final session = await _authDataSource.verifyOtp(
        challengeId: widget.challengeId,
        otp: otp,
      );

      await SecureStorage().saveAccessToken(
        session.accessToken,
      );

      if (!mounted) return;

      Navigator.pop(context, true);
    } on DioException catch (error) {
      final message = error.response?.data is Map
          ? error.response?.data['message']?.toString()
          : null;

      _showMessage(
        message ?? 'Tidak dapat menghubungi server.',
      );
    } catch (error) {
      _showMessage(
        error.toString().replaceFirst('Exception: ', ''),
      );
    } finally {
      if (mounted) {
        setState(() => _isVerifying = false);
      }
    }
  }

  Future<void> _resendOtp() async {
    setState(() => _isResending = true);

    try {
      await _authDataSource.requestOtp(
        challengeId: widget.challengeId,
      );

      if (!mounted) return;

      setState(() {
        _remainingSeconds = 300;
      });

      _startTimer();

      _showMessage('Kode OTP berhasil dikirim ulang.');
    } on DioException catch (error) {
      final message = error.response?.data is Map
          ? error.response?.data['message']?.toString()
          : null;

      );
    } catch (error) {
      _showMessage(error.toString());
    } finally {
      if (mounted) {
        setState(() => _isResending = false);
      }
    }

import 'package:flutter_secure_storage/flutter_secure_storage.dart';

class SecureStorage {
  SecureStorage() : _storage = const FlutterSecureStorage();

  final FlutterSecureStorage _storage;

  static const String _accessTokenKey = 'access_token';

  Future<void> saveAccessToken(String token) async {
    await _storage.write(
      key: _accessTokenKey,
      value: token,
    );
  }

  Future<String?> getAccessToken() async {
    return _storage.read(key: _accessTokenKey);
  }

  Future<void> deleteAccessToken() async {
    await _storage.delete(key: _accessTokenKey);
  }
}
